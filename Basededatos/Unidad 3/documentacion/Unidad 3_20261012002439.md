    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    ciudad VARCHAR(100) NOT NULL
);

CREATE TABLE empleados (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    apellidos VARCHAR(150) NOT NULL,
    email VARCHAR(150),
    salario DECIMAL(10,2),
    fecha_contratacion DATE,
    departamento_id INT,
    jefe_id INT,
    FOREIGN KEY (departamento_id) REFERENCES departamentos(id),
    FOREIGN KEY (jefe_id) REFERENCES empleados(id)
);

CREATE TABLE clientes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    ciudad VARCHAR(100),
    pais VARCHAR(100) DEFAULT 'España',
    email VARCHAR(150),
    fecha_alta DATE
);

CREATE TABLE productos (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    categoria VARCHAR(100),
    precio DECIMAL(10,2),
    stock INT
);

CREATE TABLE ventas (
    id INT AUTO_INCREMENT PRIMARY KEY,
    cliente_id INT NOT NULL,
    empleado_id INT,
    producto_id INT NOT NULL,
    cantidad INT NOT NULL,
    fecha DATE NOT NULL,
    FOREIGN KEY (cliente_id) REFERENCES clientes(id),
    FOREIGN KEY (empleado_id) REFERENCES empleados(id),
    FOREIGN KEY (producto_id) REFERENCES productos(id)
);

**10-consultas where.sql**
```sql
SELECT nombre, apellidos, posicion, goles 
FROM jugadores 
WHERE posicion = 'Delantero';

SELECT nombre AS 'Equipo', victorias 
FROM club 
WHERE victorias > 20;

SELECT 
club.nombre AS 'Equipo',
SUM(jugadores.goles) AS 'Goles totales de los delanteros'
FROM jugadores
LEFT JOIN club ON jugadores.id_club = club.id_club
WHERE jugadores.posicion = 'Delantero'
GROUP BY club.nombre
HAVING SUM(jugadores.goles) > 10
ORDER BY 'Goles totales de los delanteros';
```

**4-operadores.sql**
```sql
SELECT 
    nombre,
    apellidos,
    goles,
    goles >= 1 AS 'Goleador'
FROM jugadores;


SELECT 
    nombre, 
    apellidos, 
    posicion,
    goles,
    goles > 10 AND posicion = 'Delantero' AS 'Killer del area'
FROM jugadores;


SELECT 
    nombre, 
    apellidos, 
    posicion,
    goles,
    goles >= 1 AND posicion = 'Defensa' AS 'Defensa con gol'
FROM jugadores;


SELECT 
nombre AS 'Equipo',
victorias,
empates,
derrotas,
(victorias * 3) + empates AS 'Puntos Totales'
FROM club
```
**5-consultas de resumen.sql**
```sql
SELECT COUNT(id_jugador) AS 'Total de jugadores' 
FROM jugadores;

SELECT SUM(goles) AS 'Total de goles' 
FROM jugadores;


SELECT 
MAX(goles) AS 'Goles del Pichichi ', 
MIN(goles) AS 'Mínimo de goles' 
FROM jugadores;

SELECT AVG(goles) AS 'Promedio de goles ' 
FROM jugadores;
```
**6-consultas agruapcion.sql**
```sql
SELECT 
    posicion, 
    COUNT(id_jugador) AS 'Número de jugadores por posición'
FROM jugadores
GROUP BY posicion;

SELECT 
    posicion, 
    SUM(goles) AS 'Goles marcados por posición'
FROM jugadores
GROUP BY posicion;
```
**7-join.sql**
```sql
SELECT 
jugadores.nombre AS 'Nombre ',
jugadores.apellidos AS 'Apellidos',
club.nombre AS 'Equipo'
FROM jugadores
LEFT JOIN club ON jugadores.id_club = club.id_club;
```
**8-subconsultas.sql**
```sql
SELECT 
nombre,
apellidos,
goles,
goles > (
    SELECT AVG(goles) FROM jugadores
    ) AS 'Superior a la media '
FROM jugadores;
```
**9-combinaciones.sql**
```sql
SELECT 
club.nombre AS 'Nombre', 
SUM(jugadores.goles) AS 'Goles totales del equipo'
FROM jugadores
LEFT JOIN club ON jugadores.id_club = club.id_club
GROUP BY club.nombre;


SELECT 
jugadores.nombre AS 'Nombre',
jugadores.apellidos AS 'Apellidos',
club.nombre AS 'Equipo',
club.victorias AS 'Victorias'
FROM jugadores
LEFT JOIN club ON jugadores.id_club = club.id_club


SELECT 
jugadores.nombre AS 'Nombre del Jugador',
jugadores.apellidos AS 'Apellidos',
club.nombre AS 'Equipo',
club.victorias AS 'Victorias de su equipo'
FROM jugadores
LEFT JOIN club ON jugadores.id_club = club.id_club
ORDER BY club.victorias DESC;
```
